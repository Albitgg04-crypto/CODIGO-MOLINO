#include #include #include #include const int pinInterruptor = 10; // Interruptor deslizante const int pinMotorDC = 13; // Motor CC (Aerogenerador) const int pinServo = 9 ; // Servomotor (Orientación de aspas) const int pinTrigger = 7; // Ultrasonidos Trigger const int pinEcho = 6; // Ultrasonidos Echo const int pinDHT = 14; // Potenciómetro azul (Simulador de temperatura) #define DHTTYPE DHT11 DHT dht(pinDHT, DHTTYPE);// Creamos el objeto del sensor real //Conectar: SDA -> Pin A4 | SCL -> Pin A5 LiquidCrystal_I2C lcd(0x27, 16, 2); Servo miServo; bool estacionEncendida = false; char comandoBluetooth; long duracion; int distancia; int posicionServo = 90; void setup() { //// Iniciar monitor y Bluetooth (Pines fijos 0 y 1) Serial.begin(9600); dht.begin(); pinMode(pinInterruptor, INPUT); pinMode(pinMotorDC, OUTPUT); pinMode(pinTrigger, OUTPUT); pinMode(pinEcho, INPUT); miServo.attach(pinServo); miServo.write(posicionServo); //iniciar LCD lcd.init(); // Inicializa la pantalla lcd.backlight(); lcd.clear(); lcd.print("Estacion Eolica"); delay(2000); lcd.clear(); } void loop() { // interruptormfisico if (digitalRead(pinInterruptor) == HIGH) { estacionEncendida = true; } else { estacionEncendida = false; } // control bluetooh/ monitor serie // Envía estas letras en mayúscula desde el monitor de Tinkercad para probar la App: if (Serial.available() > 0) { comandoBluetooth = Serial.read(); if (comandoBluetooth == 'A') { // A Enciende la estación desde el móvil estacionEncendida = true; } else if (comandoBluetooth == 'D') { // D Apaga la estación desde el móvil estacionEncendida = false; } else if (comandoBluetooth == 'I') { // I Orienta el servo a la izquierda posicionServo = constrain(posicionServo - 15, 0, 180); miServo.write(posicionServo); } else if (comandoBluetooth == 'R') { // R Orienta el servo a la derecha posicionServo = constrain(posicionServo + 15, 0, 180); miServo.write(posicionServo); } } if (estacionEncendida) { // Arrancar el aerogenerador digitalWrite(pinMotorDC, HIGH); // lanza pulso ultrasónico para medir obstáculos digitalWrite(pinTrigger, LOW); delayMicroseconds(2); digitalWrite(pinTrigger, HIGH); delayMicroseconds(10); digitalWrite(pinTrigger, LOW); duracion = pulseIn(pinEcho, HIGH); distancia = duracion * 0.034 / 2; int temperaturaReal = dht.readTemperature(); // Lee la temperatura en ºC int humedadReal = dht.readHumidity(); // Lee la humedad relativa % // cumprueba si la lectura falló if (isnan(temperaturaReal) || isnan(humedadReal)) { Serial.println("¡Fallo al leer del sensor DHT!"); temperaturaReal = 0; // Valores por defecto si falla humedadReal = 0; } // Alarma Monitor Serie si obstáculos a menos de 20 cm if (distancia > 0 && distancia < 20) { Serial.println("¡ALERTA! Obstaculo aereo detectado."); } // Muestra lecturas tiempo real lcd.setCursor(0, 0); lcd.print("Dist: "); lcd.print(distancia); lcd.print("cm "); lcd.setCursor(0, 1); lcd.print("T:"); lcd.print(temperaturaReal); lcd.print("C H:"); lcd.print(humedadReal); lcd.print("% "); // Enviar los datos App Inventor Serial.print(temperaturaReal); Serial.print(","); Serial.print(humedadReal); Serial.print(","); Serial.println(distancia); } else { // Si la estación está en OFF digitalWrite(pinMotorDC, LOW); lcd.clear(); lcd.setCursor(0, 0); lcd.print("Sistema: OFF"); } delay(400); }
